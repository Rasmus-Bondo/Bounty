  <action>
Create a MovementEvents folder under game.ReplicatedStorage.Events with two RemoteEvents: SprintEvent and CrouchEvent.

Create MovementService ModuleScript at game.ServerScriptService.RoundSystem.Modules.MovementService with:

Constants:
- WALK_SPEED = 16 (normal walk)
- SPRINT_SPEED = 24 (per D-13, hold to sprint)
- CROUCH_SPEED = 8 (50% of walk speed, Claude's discretion)
- MAX_STAMINA = 100
- DRAIN_RATE = 100 / 6 (per D-14: ~6 seconds to drain fully)
- REGEN_RATE = 100 / 3.5 (per D-15: ~3.5 seconds to regen fully)
- REGEN_DELAY = 2 (per D-15: 2 seconds after sprint stops before regen begins)
- CROUCH_CAMERA_OFFSET = -1.5 (lower camera by 1.5 studs when crouching)

Module structure (following existing service pattern):
```lua
local MovementService = {}

local Players = game:GetService("Players")
local RunService = game:GetService("RunService")

-- Get RemoteEvents
local events = game.ReplicatedStorage.Events.MovementEvents
local sprintEvent = events.SprintEvent
local crouchEvent = events.CrouchEvent

function MovementService.Init()
    -- Set initial attributes for all current players
    for _, player in Players:GetPlayers() do
        MovementService.initPlayer(player)
    end
    Players.PlayerAdded:Connect(MovementService.initPlayer)
    Players.PlayerRemoving:Connect(MovementService.cleanupPlayer)

    -- Bind RemoteEvent handlers
    sprintEvent.OnServerEvent:Connect(function(player, isSprinting)
        MovementService.handleSprint(player, isSprinting)
    end)
    crouchEvent.OnServerEvent:Connect(function(player, isCrouching)
        MovementService.handleCrouch(player, isCrouching)
    end)

    -- Heartbeat tick for stamina drain/regen
    RunService.Heartbeat:Connect(function(dt)
        MovementService.tickStamina(dt)
    end)
end

function MovementService.initPlayer(player)
    player:SetAttribute("Stamina", MAX_STAMINA)
    player:SetAttribute("IsSprinting", false)
    player:SetAttribute("IsCrouching", false)
    player:SetAttribute("LastSprintStop", 0)
end

function MovementService.cleanupPlayer(player)
    -- No cleanup needed, attributes go with player
end

function MovementService.handleSprint(player, isSprinting)
    -- Type-check isSprinting (must be boolean)
    if typeof(isSprinting) ~= "boolean" then return end
    local character = player.Character
    if not character then return end
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    if not humanoid then return end

    if isSprinting then
        local stamina = player:GetAttribute("Stamina") or 0
        if stamina <= 0 then return end -- Can't sprint with no stamina
        -- Cancel crouch if crouching
        if player:GetAttribute("IsCrouching") then
            MovementService.handleCrouch(player, false)
        end
        player:SetAttribute("IsSprinting", true)
        humanoid.WalkSpeed = SPRINT_SPEED
    else
        player:SetAttribute("IsSprinting", false)
        player:SetAttribute("LastSprintStop", workspace:GetServerTimeNow())
        -- Restore to crouch or walk speed
        if player:GetAttribute("IsCrouching") then
            humanoid.WalkSpeed = CROUCH_SPEED
        else
            humanoid.WalkSpeed = WALK_SPEED
        end
    end
end

function MovementService.handleCrouch(player, isCrouching)
    if typeof(isCrouching) ~= "boolean" then return end
    local character = player.Character
    if not character then return end
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    if not humanoid then return end

    if isCrouching then
        -- Cancel sprint if sprinting
        if player:GetAttribute("IsSprinting") then
            MovementService.handleSprint(player, false)
        end
        player:SetAttribute("IsCrouching", true)
        humanoid.WalkSpeed = CROUCH_SPEED
        humanoid.CameraOffset = Vector3.new(0, CROUCH_CAMERA_OFFSET, 0)
    else
        player:SetAttribute("IsCrouching", false)
        humanoid.CameraOffset = Vector3.new(0, 0, 0)
        if player:GetAttribute("IsSprinting") then
            humanoid.WalkSpeed = SPRINT_SPEED
        else
            humanoid.WalkSpeed = WALK_SPEED
        end
    end
end

function MovementService.tickStamina(dt)
    for _, player in Players:GetPlayers() do
        local stamina = player:GetAttribute("Stamina") or MAX_STAMINA
        local isSprinting = player:GetAttribute("IsSprinting")

        if isSprinting then
            stamina = math.max(0, stamina - DRAIN_RATE * dt)
            if stamina <= 0 then
                -- Force stop sprint
                MovementService.handleSprint(player, false)
                stamina = 0
            end
        else
            local lastSprintStop = player:GetAttribute("LastSprintStop") or 0
            local now = workspace:GetServerTimeNow()
            if now - lastSprintStop >= REGEN_DELAY then
                stamina = math.min(MAX_STAMINA, stamina + REGEN_RATE * dt)
            end
        end
        player:SetAttribute("Stamina", stamina)
    end
end

return MovementService
```

```lua
local Players = game:GetService("Players")
local RunService = game:GetService("RunService")

local player = Players.LocalPlayer
local playerGui = player:WaitForChild("PlayerGui")
local healthGui = playerGui:WaitForChild("HealthDisplayGui")
local canvas = healthGui:WaitForChild("Canvas")
local staminaBar = canvas:WaitForChild("StaminaBar")
local fill = staminaBar:WaitForChild("Fill")

local displayedStamina = 100
local MAX_STAMINA = 100
local LERP_SPEED = 10 -- interpolation speed factor

RunService.RenderStepped:Connect(function(dt)
    local targetStamina = player:GetAttribute("Stamina") or MAX_STAMINA
    -- Smooth interpolation to avoid jerky updates from network replication
    displayedStamina = displayedStamina + (targetStamina - displayedStamina) * math.min(1, dt * LERP_SPEED)
    fill.Size = UDim2.new(math.clamp(displayedStamina / MAX_STAMINA, 0, 1), 0, 1, 0)
end)
```

The stamina bar should be visible at all times during gameplay. It reads the Stamina attribute set by MovementService on the server and smoothly interpolates the visual fill to avoid network jitter (per Research Pitfall 4).
