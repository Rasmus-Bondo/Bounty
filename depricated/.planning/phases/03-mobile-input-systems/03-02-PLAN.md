This script:
1. Detects platform via UserInputService.TouchEnabled (checked once at startup per Research anti-pattern advice)
2. On mobile: enables MobileControlsGui, binds touch button handlers
3. On all platforms: binds keyboard inputs via ContextActionService (createTouchButton=false since we use custom GUI per Research recommendation)
4. Handles camera swipe on mobile right-side (per D-08)

```lua
local Players = game:GetService("Players")
local UserInputService = game:GetService("UserInputService")
local ContextActionService = game:GetService("ContextActionService")

local player = Players.LocalPlayer
local playerGui = player:WaitForChild("PlayerGui")

local isMobile = UserInputService.TouchEnabled

-- Get RemoteEvents
local events = game.ReplicatedStorage:WaitForChild("Events")
local movementEvents = events:WaitForChild("MovementEvents")
local sprintEvent = movementEvents:WaitForChild("SprintEvent")
local crouchEvent = movementEvents:WaitForChild("CrouchEvent")

-- State
local isCrouching = false

-- === KEYBOARD BINDINGS (all platforms) ===

-- Sprint: hold LeftShift (per D-13: hold to sprint)
local function onSprintAction(actionName, inputState, inputObject)
    if inputState == Enum.UserInputState.Begin then
        sprintEvent:FireServer(true)
    elseif inputState == Enum.UserInputState.End then
        sprintEvent:FireServer(false)
    end
    return Enum.ContextActionResult.Sink
end

-- Crouch: toggle LeftControl
local function onCrouchAction(actionName, inputState, inputObject)
    if inputState == Enum.UserInputState.Begin then
        isCrouching = not isCrouching
        crouchEvent:FireServer(isCrouching)
    end
    return Enum.ContextActionResult.Sink
end

-- Bind keyboard actions (createTouchButton = false, we use custom GUI)
ContextActionService:BindAction("Sprint", onSprintAction, false, Enum.KeyCode.LeftShift)
ContextActionService:BindAction("Crouch", onCrouchAction, false, Enum.KeyCode.LeftControl)

-- === MOBILE TOUCH BUTTONS ===

if isMobile then
    local mobileGui = playerGui:WaitForChild("MobileControlsGui")
    mobileGui.Enabled = true

    local actionFrame = mobileGui:WaitForChild("ActionFrame")
    local shootBtn = actionFrame:WaitForChild("ShootButton")
    local reloadBtn = actionFrame:WaitForChild("ReloadButton")
    local sprintBtn = actionFrame:WaitForChild("SprintButton")
    local crouchBtn = actionFrame:WaitForChild("CrouchButton")

    -- Shoot button: fires the gun
    -- NOTE: The gun is a Tool with GunClient handling Tool.Activated.
    -- On mobile, the shoot button should call Tool:Activate() on the equipped tool.
    -- This triggers the same Tool.Activated path as clicking on PC.
    shootBtn.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.Touch then
            local character = player.Character
            if character then
                local tool = character:FindFirstChildOfClass("Tool")
                if tool then
                    tool:Activate()
                end
            end
        end
    end)

    -- Reload button: fires reload event
    -- Read GunClient to find exact reload event name and pattern
    reloadBtn.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.Touch then
            -- Fire reload through the same mechanism as keyboard R
            -- Check GunClient for reload binding pattern and replicate
            local combatEvents = events:FindFirstChild("CombatEvents")
            if combatEvents then
                local reloadEvent = combatEvents:FindFirstChild("ReloadEvent")
                if reloadEvent then
                    reloadEvent:FireServer()
                end
            end
        end
    end)

    -- Sprint button: hold to sprint (per D-13)
    sprintBtn.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.Touch then
            sprintEvent:FireServer(true)
        end
    end)
    sprintBtn.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.Touch then
            sprintEvent:FireServer(false)
        end
    end)

    -- Crouch button: toggle (tap to crouch, tap again to stand)
    crouchBtn.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.Touch then
            isCrouching = not isCrouching
            crouchEvent:FireServer(isCrouching)
        end
    end)
end
```
