-- After: local raycastResult = workspace:Raycast(origin, direction * range, raycastParams)
if raycastResult then
    local hit = raycastResult.Instance
    
    -- Check if hit is a window (name-based or tag-based per DestructibleService implementation)
    if hit.Name == "Window" and not DestructibleService.isWindowDestroyed(hit) then
        -- Break the window first
        DestructibleService.breakWindow(hit)
        
        -- Continue raycast from same origin, past the window  
        -- Exclude the window from second raycast
        raycastParams.FilterDescendantsInstances = {shooter.Character, hit}
        local continuedRaycast = workspace:Raycast(origin, direction * range, raycastParams)
        
        if continuedRaycast then  
            -- Process the continued hit (could be player behind window)
            raycastResult = continuedRaycast
            hit = continuedRaycast.Instance
        else
            -- No further hits, window was only target
            return  
        end
        
        -- Reset filter for next shot
        raycastParams.FilterDescendantsInstances = {shooter.Character}
    end
    
    -- Continue with existing hit processing logic for players
    local hitPlayer = Players:GetPlayerFromCharacter(hit.Parent)
    if hitPlayer then
        -- Existing damage logic continues unchanged
    end
end
