**StatMenuScript (LocalScript):**

```lua
local Players = game:GetService("Players")
local UIS = game:GetService("UserInputService")
local player = Players.LocalPlayer

-- References
local mainFrame = script.Parent:WaitForChild("MainFrame")
local myStatsPanel = mainFrame.ContentFrame.MyStatsPanel
local scoreboardPanel = mainFrame.ContentFrame.ScoreboardPanel
local postGamePanel = mainFrame.ContentFrame.PostGamePanel
local myStatsTab = mainFrame.TabBar.MyStatsTab
local scoreboardTab = mainFrame.TabBar.ScoreboardTab

-- ProgressionEvents
local events = game.ReplicatedStorage:WaitForChild("Events"):WaitForChild("ProgressionEvents")
local requestStats = events:WaitForChild("RequestStats")
local statsResponse = events:WaitForChild("StatsResponse")

local isOpen = false
local activeTab = "mystats"

-- Stat label references
local labels = {
    Cash = myStatsPanel.ProgressionSection.CashLabel,
    Level = myStatsPanel.ProgressionSection.LevelLabel,
    XP = myStatsPanel.ProgressionSection.XPLabel,
    Kills = myStatsPanel.SessionSection.KillsLabel,
    Deaths = myStatsPanel.SessionSection.DeathsLabel,
    KD = myStatsPanel.SessionSection.KDLabel,
    TimeAlive = myStatsPanel.SessionSection.TimeAliveLabel,
    TimeDead = myStatsPanel.SessionSection.TimeDeadLabel,
    LifetimeKills = myStatsPanel.LifetimeSection.LifetimeKillsLabel,
    LifetimeDeaths = myStatsPanel.LifetimeSection.LifetimeDeathsLabel,
    LifetimeKD = myStatsPanel.LifetimeSection.LifetimeKDLabel,
    Playtime = myStatsPanel.LifetimeSection.PlaytimeLabel,
}

-- Helper: format time in seconds to human-readable
local function formatTime(seconds)
    seconds = math.floor(seconds)
    if seconds < 3600 then
        return string.format("%d:%02d", math.floor(seconds / 60), seconds % 60)
    else
        local h = math.floor(seconds / 3600)
        local m = math.floor((seconds % 3600) / 60)
        return string.format("%dh %dm", h, m)
    end
end

-- Helper: format K/D ratio (handle 0 deaths)
local function formatKD(kills, deaths)
    if deaths == 0 then
        return tostring(kills) .. ".00"
    end
    return string.format("%.2f", kills / deaths)
end

-- Populate My Stats from server response
local function populateStats(data)
    labels.Cash.Text = "Cash: $" .. tostring(data.Cash)
    labels.Level.Text = "Level: " .. tostring(data.Level)
    labels.XP.Text = "XP: " .. tostring(data.XP % 1000) .. " / 1000"
    labels.Kills.Text = "Kills: " .. tostring(data.SessionKills)
    labels.Deaths.Text = "Deaths: " .. tostring(data.SessionDeaths)
    labels.KD.Text = "K/D: " .. formatKD(data.SessionKills, data.SessionDeaths)
    labels.TimeAlive.Text = "Time Alive: " .. formatTime(data.TimeAlive)
    labels.TimeDead.Text = "Time Dead: " .. formatTime(data.TimeDead)
    labels.LifetimeKills.Text = "Kills: " .. tostring(data.LifetimeKills)
    labels.LifetimeDeaths.Text = "Deaths: " .. tostring(data.LifetimeDeaths)
    labels.LifetimeKD.Text = "K/D: " .. formatKD(data.LifetimeKills, data.LifetimeDeaths)
    labels.Playtime.Text = "Playtime: " .. formatTime(data.TotalPlaytime)
end

-- Populate Scoreboard (STAT-03)
-- Reads SessionKills and SessionDeaths from player attributes (set by DataService in Plan 01)
local function populateScoreboard()
    -- Clear existing player rows (keep HeaderRow)
    for _, child in scoreboardPanel:GetChildren() do
        if child:IsA("Frame") and child.Name ~= "HeaderRow" then
            child:Destroy()
        end
    end

    local order = 1
    for _, p in Players:GetPlayers() do
        local row = Instance.new("Frame")
        row.Name = "PlayerRow_" .. p.Name
        row.Size = UDim2.new(1, 0, 0, 30)
        row.BackgroundTransparency = (order % 2 == 0) and 0.9 or 1
        row.BackgroundColor3 = Color3.fromRGB(50, 50, 50)
        row.LayoutOrder = order

        local nameLabel = Instance.new("TextLabel")
        nameLabel.Size = UDim2.new(0.5, 0, 1, 0)
        nameLabel.Text = p.Name
        nameLabel.Font = Enum.Font.GothamMedium
        nameLabel.TextSize = 14
        nameLabel.TextColor3 = Color3.fromRGB(220, 220, 220)
        nameLabel.TextXAlignment = Enum.TextXAlignment.Left
        nameLabel.BackgroundTransparency = 1
        nameLabel.Parent = row

        -- Read from player attributes set by DataService.incrementSessionStat (Plan 01)
        local killsVal = p:GetAttribute("SessionKills") or 0
        local deathsVal = p:GetAttribute("SessionDeaths") or 0

        local killsLabel = Instance.new("TextLabel")
        killsLabel.Size = UDim2.new(0.25, 0, 1, 0)
        killsLabel.Position = UDim2.new(0.5, 0, 0, 0)
        killsLabel.Text = tostring(killsVal)
        killsLabel.Font = Enum.Font.GothamMedium
        killsLabel.TextSize = 14
        killsLabel.TextColor3 = Color3.fromRGB(220, 220, 220)
        killsLabel.BackgroundTransparency = 1
        killsLabel.Parent = row

        local deathsLabel = Instance.new("TextLabel")
        deathsLabel.Size = UDim2.new(0.25, 0, 1, 0)
        deathsLabel.Position = UDim2.new(0.75, 0, 0, 0)
        deathsLabel.Text = tostring(deathsVal)
        deathsLabel.Font = Enum.Font.GothamMedium
        deathsLabel.TextSize = 14
        deathsLabel.TextColor3 = Color3.fromRGB(220, 220, 220)
        deathsLabel.BackgroundTransparency = 1
        deathsLabel.Parent = row

        row.Parent = scoreboardPanel
        order += 1
    end
end

-- Tab switching
local function switchTab(tab)
    activeTab = tab
    myStatsPanel.Visible = (tab == "mystats")
    scoreboardPanel.Visible = (tab == "scoreboard")
    myStatsTab.BackgroundColor3 = (tab == "mystats") and Color3.fromRGB(60, 60, 60) or Color3.fromRGB(35, 35, 35)
    scoreboardTab.BackgroundColor3 = (tab == "scoreboard") and Color3.fromRGB(60, 60, 60) or Color3.fromRGB(35, 35, 35)
end

myStatsTab.Activated:Connect(function() switchTab("mystats") end)
scoreboardTab.Activated:Connect(function() switchTab("scoreboard") end)

-- Toggle menu
local function toggleMenu()
    isOpen = not isOpen
    mainFrame.Visible = isOpen
    if isOpen then
        requestStats:FireServer()
        if activeTab == "scoreboard" then
            populateScoreboard()
        end
    end
end

-- Stats response handler
statsResponse.OnClientEvent:Connect(function(data)
    populateStats(data)
end)

-- PC: Tab key toggle (per D-14)
UIS.InputBegan:Connect(function(input, processed)
    if processed then return end
    if input.KeyCode == Enum.KeyCode.Tab then
        toggleMenu()
    end
end)

-- Mobile button handler will be connected by InputController (Task 2)
-- Export toggle function for external callers
script.Parent.MainFrame:SetAttribute("ToggleFunction", "toggleMenu")
```

**NOTE:** This plan does NOT modify DataService. The SetAttribute calls for SessionKills/SessionDeaths are already handled in Plan 01's DataService.incrementSessionStat function. The scoreboard reads these attributes directly from player objects -- no cross-plan modification needed.
