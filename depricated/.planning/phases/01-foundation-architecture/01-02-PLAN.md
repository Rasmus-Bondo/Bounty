Replace RoleService via `mcp__robloxstudio__set_script_source` instancePath="game.ServerScriptService.RoundSystem.Modules.RoleService":
```lua
local Players = game:GetService("Players")

local RoleService = {}

function RoleService.clearRoles()
	for _, player in ipairs(Players:GetPlayers()) do
		player:SetAttribute("Role", nil)
	end
end

function RoleService.setAllBandits()
	for _, player in ipairs(Players:GetPlayers()) do
		player:SetAttribute("Role", "Bandit")
	end
end

function RoleService.eliminatePlayer(player)
	if not player then return end
	if player:GetAttribute("Role") ~= "Bandit" then return end

	player:SetAttribute("Role", "Eliminated")

	task.defer(function()
		if player and player.Parent then
			player:LoadCharacter()
		end
	end)
end

return RoleService
```
Key changes: Removed chooseBandit, setRoles, countAliveTownsfolk, eliminateTownsfolkImmediately. Added setAllBandits. Simplified eliminatePlayer.
