local Players = game:GetService("Players")
local RunService = game:GetService("RunService")
local player = Players.LocalPlayer

local Events = game.ReplicatedStorage:WaitForChild("Events")

local LobbyGui = script.Parent
local JoinButton = LobbyGui:WaitForChild("JoinButton")

local TimerLabel = InfoFrame:WaitForChild("TimerLabel")
local PlayerCountLabel = InfoFrame:WaitForChild("PlayerCountLabel")

-- Button toggle behavior (per D-01)
local function updateButton()
    local optedIn = player:GetAttribute("OptedIn")
    if optedIn then
        JoinButton.Text = "Leave Queue"
        JoinButton.BackgroundColor3 = Color3.fromRGB(180, 60, 60)  -- red
        -- Also update stroke color
        local stroke = JoinButton:FindFirstChildOfClass("UIStroke")
        if stroke then stroke.Color = Color3.fromRGB(140, 40, 40) end
    else
        JoinButton.Text = "Join Round"
        JoinButton.BackgroundColor3 = Color3.fromRGB(60, 140, 60)  -- green
        local stroke = JoinButton:FindFirstChildOfClass("UIStroke")
        if stroke then stroke.Color = Color3.fromRGB(40, 100, 40) end
    end
end

JoinButton.Activated:Connect(function()
    local optedIn = player:GetAttribute("OptedIn")
    if optedIn then
        RoundEvent:FireServer("LeaveRound")
    else
        RoundEvent:FireServer("JoinRound")
    end
end)

player:GetAttributeChangedSignal("OptedIn"):Connect(updateButton)
updateButton()

-- Info display updates (per D-04)
-- Read round info from shared attributes on a well-known object
-- The server will set attributes on a Configuration object for round state
-- Alternative: use RoundEvent to periodically update clients
-- Simpler approach: poll leaderstats / player count each frame

local function updateInfo()
    -- Player count: count players with OptedIn = true
    local count = 0
    for _, p in ipairs(Players:GetPlayers()) do
        if p:GetAttribute("OptedIn") then
            count = count + 1
        end
    end
    PlayerCountLabel.Text = "Players: " .. count
end

-- Update info every second (lightweight poll)
task.spawn(function()
    while true do
        updateInfo()
        task.wait(1)
    end
end)

**PostGameScript** (LocalScript, child of PostGameGui):
```lua
local Players = game:GetService("Players")
local player = Players.LocalPlayer
local Events = game.ReplicatedStorage:WaitForChild("Events")
local PostGameStats = Events:WaitForChild("PostGameStats")

local PostGameGui = script.Parent
local Overlay = PostGameGui:WaitForChild("Overlay")
local PlayerList = Overlay.LeaderboardFrame:WaitForChild("PlayerList")

local function formatKD(kills, deaths)
    if deaths == 0 then
        return string.format("%.1f", kills)
    end
    return string.format("%.2f", kills / deaths)
end

local function clearPlayerList()
    for _, child in PlayerList:GetChildren() do
        if child:IsA("Frame") then
            child:Destroy()
        end
    end
end

local function createPlayerRow(rank, data, isLocalPlayer)
    local row = Instance.new("Frame")
    row.Size = UDim2.new(1, 0, 0, 30)
    row.BackgroundColor3 = isLocalPlayer and Color3.fromRGB(60, 50, 30) or Color3.fromRGB(0, 0, 0)
    row.BackgroundTransparency = isLocalPlayer and 0.3 or 0.8
    row.LayoutOrder = rank
    row.Name = "Row_" .. rank

    local corner = Instance.new("UICorner")
    corner.CornerRadius = UDim.new(0, 4)
    corner.Parent = row

    local textColor = isLocalPlayer and Color3.new(1, 0.9, 0.5) or Color3.fromRGB(220, 220, 220)
    local font = isLocalPlayer and Enum.Font.GothamBold or Enum.Font.Gotham

    -- Rank
    local rankLabel = Instance.new("TextLabel")
    rankLabel.Size = UDim2.new(0.1, 0, 1, 0)
    rankLabel.Position = UDim2.new(0, 0, 0, 0)
    rankLabel.BackgroundTransparency = 1
    rankLabel.Text = "#" .. rank
    rankLabel.TextColor3 = textColor
    rankLabel.Font = font
    rankLabel.TextSize = 16
    rankLabel.Parent = row

    -- Name
    local nameLabel = Instance.new("TextLabel")
    nameLabel.Size = UDim2.new(0.35, 0, 1, 0)
    nameLabel.Position = UDim2.new(0.1, 0, 0, 0)
    nameLabel.BackgroundTransparency = 1
    nameLabel.Text = data.name
    nameLabel.TextColor3 = textColor
    nameLabel.Font = font
    nameLabel.TextSize = 16
    nameLabel.TextXAlignment = Enum.TextXAlignment.Left
    nameLabel.Parent = row

    -- Kills
    local killsLabel = Instance.new("TextLabel")
    killsLabel.Size = UDim2.new(0.18, 0, 1, 0)
    killsLabel.Position = UDim2.new(0.45, 0, 0, 0)
    killsLabel.BackgroundTransparency = 1
    killsLabel.Text = tostring(data.kills)
    killsLabel.TextColor3 = textColor
    killsLabel.Font = font
    killsLabel.TextSize = 16
    killsLabel.Parent = row

    -- Deaths
    local deathsLabel = Instance.new("TextLabel")
    deathsLabel.Size = UDim2.new(0.18, 0, 1, 0)
    deathsLabel.Position = UDim2.new(0.63, 0, 0, 0)
    deathsLabel.BackgroundTransparency = 1
    deathsLabel.Text = tostring(data.deaths)
    deathsLabel.TextColor3 = textColor
    deathsLabel.Font = font
    deathsLabel.TextSize = 16
    deathsLabel.Parent = row

    -- K/D
    local kdLabel = Instance.new("TextLabel")
    kdLabel.Size = UDim2.new(0.19, 0, 1, 0)
    kdLabel.Position = UDim2.new(0.81, 0, 0, 0)
    kdLabel.BackgroundTransparency = 1
    kdLabel.Text = formatKD(data.kills, data.deaths)
    kdLabel.TextColor3 = textColor
    kdLabel.Font = font
    kdLabel.TextSize = 16
    kdLabel.Parent = row

    row.Parent = PlayerList
    return row
end

PostGameStats.OnClientEvent:Connect(function(stats)
    clearPlayerList()

    -- Show top 5 (per D-09)
    local localPlayerShown = false
    local displayCount = math.min(#stats, 5)

    for i = 1, displayCount do
        local isLocal = stats[i].name == player.DisplayName
        if isLocal then localPlayerShown = true end
        createPlayerRow(i, stats[i], isLocal)
    end

    -- If local player not in top 5, show them at bottom with separator (per D-09)
    if not localPlayerShown then
        for i, data in ipairs(stats) do
            if data.name == player.DisplayName then
                -- Add separator
                local sep = Instance.new("Frame")
                sep.Size = UDim2.new(0.8, 0, 0, 1)
                sep.Position = UDim2.new(0.1, 0, 0, 0)
                sep.BackgroundColor3 = Color3.fromRGB(100, 100, 100)
                sep.LayoutOrder = 6
                sep.Name = "Separator"
                sep.Parent = PlayerList

                createPlayerRow(i, data, true)
                break
            end
        end
    end

    -- Update canvas size
    local layout = PlayerList:FindFirstChildOfClass("UIListLayout")
    if layout then
        PlayerList.CanvasSize = UDim2.new(0, 0, 0, layout.AbsoluteContentSize.Y)
    end

    -- Show overlay
    Overlay.Visible = true

    -- Auto-hide after intermission (10 seconds per D-10)
    task.delay(10, function()
        Overlay.Visible = false
    end)
end)
```
