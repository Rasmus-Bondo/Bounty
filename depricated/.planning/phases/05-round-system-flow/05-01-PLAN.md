**LobbyService ModuleScript** (create at game.ServerScriptService.RoundSystem.Modules.LobbyService):
```lua
local LobbyService = {}

local Players = game:GetService("Players")
local State = require(script.Parent.State)
local Config = require(script.Parent.Config)
local SpawnService = require(script.Parent.SpawnService)
local BanditService = require(script.Parent.BanditService)

function LobbyService.Initialize()
    -- Hook PlayerRemoving to clean up opt-in state (Pitfall 5)
    Players.PlayerRemoving:Connect(function(player)
        LobbyService.optOut(player)
    end)

    -- Listen for RoundEvent from clients
    local Events = game.ReplicatedStorage:WaitForChild("Events")
    local RoundEvent = Events:WaitForChild("RoundEvent")
    RoundEvent.OnServerEvent:Connect(function(player, action)
        if action == "JoinRound" then
            LobbyService.optIn(player)
        elseif action == "LeaveRound" then
            LobbyService.optOut(player)
        end
    end)

    print("[LobbyService] Initialized")
end

function LobbyService.optIn(player)
    player:SetAttribute("OptedIn", true)
    State.optedInPlayers[player] = true
    print("[LobbyService] " .. player.Name .. " opted in")

    -- If round is active, immediately spawn into round (per D-05)
    if State.roundActive then
        SpawnService.sendPlayerToRoundSpawn(player)
        BanditService.giveBanditGun(player)
        -- InvulnService will be called from hookCharacter after Plan 02 wires it
    end
end

function LobbyService.optOut(player)
    player:SetAttribute("OptedIn", false)
    State.optedInPlayers[player] = nil
    print("[LobbyService] " .. player.Name .. " opted out")
end

function LobbyService.isOptedIn(player)
    return player:GetAttribute("OptedIn") == true
end

function LobbyService.getOptedInPlayers()
    local opted = {}
    for player, _ in pairs(State.optedInPlayers) do
        if player.Parent then -- still connected
            table.insert(opted, player)
        end
    end
    return opted
end

function LobbyService.getOptedInCount()
    local count = 0
    for player, _ in pairs(State.optedInPlayers) do
        if player.Parent then
            count = count + 1
        end
    end
    return count
end

function LobbyService.resetAllOptIns()
    -- Do NOT reset opt-ins between rounds (per Claude's discretion: persist across rounds for fluid experience)
    -- Only called if explicitly needed
    for player, _ in pairs(State.optedInPlayers) do
        player:SetAttribute("OptedIn", false)
    end
    State.optedInPlayers = {}
end

return LobbyService
```

**InvulnService ModuleScript** (create at game.ServerScriptService.RoundSystem.Modules.InvulnService):

Per D-13 (10-second duration), D-14 (flickering transparent), D-15 (accelerating fade), D-16 (50% transparency), D-17 (cancel on firing).

Server-side implementation for damage protection. Visual flicker handled server-side via task.spawn loop so ALL players see the flicker (Research Open Question 3 recommendation).

```lua
local InvulnService = {}

local Config = require(script.Parent.Config)

-- Track active invuln threads so they can be cancelled
local activeInvulns = {} -- {[player] = true}

function InvulnService.applyInvulnerability(player)
    local character = player.Character
    if not character then return end

    local humanoid = character:WaitForChild("Humanoid", 5)
    if not humanoid then return end

    -- Set attribute FIRST so GunServer can check it
    player:SetAttribute("Invulnerable", true)
    activeInvulns[player] = true

    -- Create invisible ForceField for damage protection (Pitfall 1: ForceField blocks TakeDamage)
    local forceField = Instance.new("ForceField")
    forceField.Visible = false
    forceField.Name = "SpawnProtection"
    forceField.Parent = character

    -- Server-side flicker loop (so all clients see it via Transparency replication)
    task.spawn(function()
        local startTime = tick()
        local duration = Config.INVULN_DURATION
        local warningStart = duration - Config.INVULN_WARNING_TIME

        while activeInvulns[player] and player.Parent and character.Parent do
            local elapsed = tick() - startTime

            -- Auto-expire after duration
            if elapsed >= duration then
                break
            end

            -- Calculate flicker interval (per D-15: accelerating in final 3s)
            local interval
            if elapsed < warningStart then
                interval = Config.INVULN_FLICKER_BASE  -- 0.3s
            else
                local progress = (elapsed - warningStart) / Config.INVULN_WARNING_TIME
                interval = Config.INVULN_FLICKER_BASE - ((Config.INVULN_FLICKER_BASE - Config.INVULN_FLICKER_FAST) * math.min(progress, 1))
            end

            -- Set character body parts to 50% transparent (per D-16)
            -- Filter to body parts only (Pitfall 2: avoid Tools/Accessories)
            for _, part in character:GetChildren() do
                if part:IsA("BasePart") then
                    part.Transparency = 0.5
                end
            end
            task.wait(interval)

            -- Set back to opaque
            for _, part in character:GetChildren() do
                if part:IsA("BasePart") then
                    part.Transparency = 0
                end
            end
            task.wait(interval)
        end

        -- Clean up: ensure fully opaque
        if character.Parent then
            for _, part in character:GetChildren() do
                if part:IsA("BasePart") then
                    part.Transparency = 0
                end
            end
        end

        -- Remove ForceField if still exists
        InvulnService.removeInvulnerability(player)
    end)

    print("[InvulnService] Applied invulnerability to " .. player.Name)
end

function InvulnService.removeInvulnerability(player)
    activeInvulns[player] = nil
    player:SetAttribute("Invulnerable", false)

    local character = player.Character
    if character then
        local forceField = character:FindFirstChild("SpawnProtection")
        if forceField then
            forceField:Destroy()
        end
        -- Ensure opaque
        for _, part in character:GetChildren() do
            if part:IsA("BasePart") then
                part.Transparency = 0
            end
        end
    end
end

function InvulnService.isInvulnerable(player)
    return player:GetAttribute("Invulnerable") == true
end

-- Cleanup on disconnect
function InvulnService.Initialize()
    game.Players.PlayerRemoving:Connect(function(player)
        activeInvulns[player] = nil
    end)
    print("[InvulnService] Initialized")
end

return InvulnService
```
